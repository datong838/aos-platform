"""Independent read-only outcome audit; never replay COMMIT or remove a database."""
import hashlib
import json
import subprocess
from datetime import datetime
from pathlib import Path

import psycopg
from psycopg.rows import dict_row

B = Path(__file__).resolve().parent
R = B.parents[1]
lab = json.loads((B / 'lab.json').read_text())
result = json.loads((B / 'native-post-finalize-expiry-r3-result.json').read_text())
assert result['pytest_exit'] == 0 and len(result['results']) == 2
observations = [json.loads(p.read_text()) for p in sorted(B.glob('observation-*.json'))]
assert len(observations) == 2 and len({r['database'] for r in observations}) == 2
reports = []
for item in observations:
    parse = datetime.fromisoformat
    assert parse(item['finalized_at']) < parse(item['authentication_deadline'])
    assert parse(item['authentication_deadline']) < parse(item['before_commit'])
    assert parse(item['before_commit']) < parse(item['source_valid_until'])
    assert item['execution_allowed'] is False and not item['semantic_decision_made']
    with psycopg.connect(host='127.0.0.1', port=5436, user='aos_app',
                         dbname=item['database'], passfile=lab['passfile'],
                         row_factory=dict_row) as conn:
        conn.execute('SET TRANSACTION ISOLATION LEVEL SERIALIZABLE READ ONLY')
        assert conn.execute('SELECT system_identifier::text AS id FROM pg_control_system()').fetchone()['id'] == lab['cluster']
        status = conn.execute('SELECT pg_xact_status(%s::xid8) AS state',
                              (item['identity']['xid'],)).fetchone()['state']
        sources = conn.execute("""SELECT p.asset_id,p.content_hash,
            p.content_hash=encode(sha256(p.source_raw),'hex') AS hash_ok,
            aos_ed25519.verify(p.signature,p.source_raw,
              decode(s.signed_facts->>'sourceSignerPublicKey','base64')) AS signature_ok,
            s.finalized,v.native_xid::text AS original_xid,v.native_pid AS original_pid,
            v.native_session AS original_session,
            v.native_xid=s.native_xid AND v.native_pid=s.native_pid
              AND v.native_session=s.native_session AS provenance_ok
            FROM aos_capacity_publication_private.source_publication p
            JOIN aos_capacity_publication_private.source_pending s ON s.attempt_id=p.attempt_id
            JOIN aos_capacity_publication_private.source_provenance v
              USING(org_id,project_id,asset_id) WHERE p.asset_id=%s""", (item['asset'],)).fetchall()
        if item['result']['outcome'] == 'committed':
            assert status == 'committed' and len(sources) == 1
            row = sources[0]
            assert all(row[k] for k in ('hash_ok','signature_ok','finalized','provenance_ok'))
            assert row['content_hash'] == item['source_sha256']
            assert row['original_xid'] == item['identity']['xid']
            assert row['original_pid'] == item['identity']['pid']
            assert row['original_session'] == 'aos_provider_capacity_issuer'
        else:
            assert status == 'aborted' and not sources
        assert conn.execute("SELECT count(*) AS n FROM pg_roles WHERE rolname IN"
            " ('aos_provider_capacity_issuer','aos_provider_capacity_occupation') AND rolcanlogin").fetchone()['n'] == 0
        for signature in ('public.prepare_capacity_source(bytea,text,text,jsonb)',
                          'public.commit_capacity_source(text,bytea,bytea)',
                          'aos_capacity_publication_private.require_current_source_installation()'):
            assert conn.execute('SELECT to_regprocedure(%s) IS NULL AS absent',
                                (signature,)).fetchone()['absent']
        reports.append(dict(database=item['database'],outcome=item['result'],
                            original_xact_state=status,sources=sources,
                            temporary_public_interfaces_removed=True,issuer_login_restored=True))
for binding in json.loads((B/'baseline-files.json').read_text()):
    assert hashlib.sha256((R/binding['path']).read_bytes()).hexdigest() == binding['sha256']
old = json.loads((R/'.evidence/gt-source-newjudge-native-lab-20261009/preflight.json').read_text())['old_containers']
for item in old:
    expected = item['started']
    if item['Id'].startswith('9cc6091c304b55'):
        expected = json.loads((R/'.evidence/current157-full-restart-20261009-010759/final-independent-verification.json').read_text())['pg_started']
    now = json.loads(subprocess.check_output(['docker','--context','colima-aos-extension','inspect',item['Id']],text=True))[0]
    canonical = lambda value: sorted(value,key=lambda x:json.dumps(x,sort_keys=True))
    assert now['Id'] == item['Id'] and now['State']['StartedAt'] == expected
    assert canonical(now['Mounts']) == canonical(item['mounts'])
    assert now['HostConfig']['PortBindings'] == item['ports']
(B/'independent-verification.json').write_text(json.dumps(dict(observations=reports,
    original_products_unchanged=True,old_resources_unchanged=True,old_resource_count=len(old),
    final_commit_authentication_semantics_decided=False,formal_execution_allowed=False),indent=2)+'\n')
print('Independent Source outcome/signature/provenance/role/resource verification PASS')
