class _RenewalCommand(AipContractModel):
    model_config = ConfigDict(frozen=True)
    domain: ClassVar[str]
    schema_version: Literal[1]
    org_id: Literal['org-org']
    project_id: Literal['dev-project']
    idempotency_key: Annotated[StrictStr, Field(
        min_length=1, max_length=120, pattern=r'^[A-Za-z0-9][A-Za-z0-9._:-]*$')]

    @field_validator('schema_version', mode='before')
    @classmethod
    def _version(cls, value):
        if type(value) is not int:
            raise ValueError('exact integer schema version required')
        return value

    @property
    def activation_eligible(self) -> Literal[False]:
        return False

    def canonical_payload(self) -> dict:
        # model_copy can bypass validators, so every usable hash reparses fields.
        checked = type(self).model_validate(self.model_dump(mode='python', by_alias=True))
        return checked.model_dump(mode='json', by_alias=True)

    @property
    def request_hash(self) -> str:
        value = self.canonical_payload()
        value.pop('idempotencyKey')
        # The authenticated actor and original key belong to the Owner's receipt
        # namespace; this hash binds the operation's nonsecret semantic input.
        return canonical_hash({'domain': self.domain, 'command': value})

def _database_identity(conn, purpose):
    if type(conn) is not psycopg.Connection or conn.closed:
        raise RenewalIdentityBlocked('RENEWAL_REAL_DATABASE_SESSION_REQUIRED')
    try:
        with conn.cursor(row_factory=dict_row) as cursor:
            row = cursor.execute("""SELECT session_user AS login,current_user AS role,
                current_setting('aos.org_id',true) AS org_id,
                current_setting('aos.project_id',true) AS project_id,
                current_schemas(true) AS schemas,
                r.rolcanlogin,r.rolsuper,r.rolbypassrls,r.rolinherit,
                r.rolcreatedb,r.rolcreaterole,r.rolreplication,
                EXISTS(SELECT 1 FROM pg_auth_members
                       WHERE roleid=r.oid OR member=r.oid) AS membership
                FROM pg_roles r WHERE r.rolname=session_user""").fetchone()
        if (not row or row['login'] != _ROLES[purpose] or row['role'] != row['login']
                or (row['org_id'], row['project_id']) != _SCOPE
                or row['rolcanlogin'] is not True
                or any(row[k] is not False for k in (
                    'rolsuper', 'rolbypassrls', 'rolinherit', 'rolcreatedb',
                    'rolcreaterole', 'rolreplication', 'membership'))
                or not row['schemas']
                or not set(row['schemas']) <= {'pg_catalog', 'public'}):
            raise RenewalIdentityBlocked('RENEWAL_DATABASE_IDENTITY_PROFILE_REQUIRED')
        return row['login']
    except RenewalIdentityBlocked:
        raise
    except Exception:
        raise RenewalIdentityBlocked('RENEWAL_DATABASE_IDENTITY_UNAVAILABLE') from None

class RenewalAuthorityStore:
    def __init__(self, conn, *, checker_binding=None):

        self._binding = checker_binding
        self._idle()

    def _idle(self):

        if (type(c) is not psycopg.Connection or c.closed or c.autocommit
                or c.info.transaction_status != TransactionStatus.IDLE
                or c.info.user not in _CALLERS.values()):
            raise RenewalStoreBlocked('RENEWAL_REAL_IDLE_DATABASE_CONNECTION_REQUIRED')

    def propose(self, command):
        return self._call(command, expected=RenewalProposalCommand)

    def decide(self, command, *, token):
        return self._call(command, expected=RenewalDecisionCommand, token=token)

    def revoke(self, command, *, token):
        return self._call(command, expected=RenewalRevocationCommand, token=token)

    def consume(self, command):
        return self._call(command, expected=RenewalConsumeCommand)
