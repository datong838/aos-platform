def _non_negative_int(value: object) -> int | None:
    if value is None or isinstance(value, bool):
        return None
    try:
        number = Decimal(str(value))
    except InvalidOperation:
        return None
    if not number.is_finite() or number < 0 or number != number.to_integral_value():
        return None
    return int(number)


class EcommerceWorkshopContentCampaignInputs:
    def __init__(
        self,
        *,
        clock: Clock | None = None,
        connect_factory: ConnectFactory | None = None,
        campaign_store: EcommerceContentCampaignAuthorityStore | None = None,
    ) -> None:
        self._clock = clock or (lambda: datetime.now(UTC))
        self._connect_factory = connect_factory or partial(
            connect_read_only, inherit_scope=False
        )
        self._campaign_store = campaign_store or EcommerceContentCampaignAuthorityStore()

    def read(self, *, org_id: str, project_id: str):
        cutoff = self._clock()
        if cutoff.utcoffset() is None:
            raise ValueError("content-campaign input clock requires timezone")
        scope = TenantScope(org_id, project_id)
        products: list[ContentCampaignProductInput] = []
        inventory: list[ContentCampaignInventoryInput] = []
        plans: list[ContentCampaignPlanInput] = []
        unavailable: list[str] = []
        try:
            with self._connect_factory() as conn:
                apply_transaction_scope(conn, scope)
                rows = conn.execute(
                    "SELECT object_type,external_id,properties,source_updated_at,payload_hash "
                    "FROM ecom_object WHERE org_id=%s AND workspace_id=%s "
                    "AND object_type IN (%s,%s) AND deleted_at IS NULL "
                    "AND source_updated_at<=%s ORDER BY object_type,source_updated_at DESC,external_id DESC LIMIT 200",
                    (*scope.key, "Product", "ProductSku", cutoff),
                ).fetchall()
            product_rows = [row for row in rows if row["object_type"] == "Product"][:100]
            sku_rows = [row for row in rows if row["object_type"] == "ProductSku"][:100]
            products = [self._product(row) for row in product_rows]
            product_by_source_id = {
                item.product_ref.resource_id.rsplit(":", 1)[-1]: item.product_ref
                for item in products
            }
            inventory = [self._inventory(row, product_by_source_id) for row in sku_rows]
        except (psycopg.Error, KeyError, TypeError, ValueError):
            unavailable.extend(["products", "inventory"])
        try:
            snapshot = self._campaign_store.read_snapshot(scope, cutoff=cutoff, limit=100)
            if snapshot.errors:
                raise ContentCampaignAuthorityReadError("campaign snapshot partial")
            plans = [self._campaign(item) for item in snapshot.campaigns]
            plans.extend(self._calendar(item) for item in snapshot.calendar_entries)
        except ContentCampaignAuthorityReadError:
            unavailable.append("plans")
        status = "blocked" if len(unavailable) == 3 else ("partial" if unavailable else "ready")
        unmatched = sum(item.product_ref is None for item in inventory)
        return WorkshopContentCampaignInputsEnvelope(
            tenant={"orgId": org_id, "projectId": project_id},
            evaluated_at=cutoff,
            data_cutoff=cutoff,
            status=status,
            products=products,
            inventory=inventory,
            plans=plans,
            ledger=ContentCampaignInputLedger(
                products=len(products), inventory=len(inventory), plans=len(plans),
                unmatched_inventory=unmatched,
            ),
            unavailable_groups=unavailable,
        )

    @staticmethod
    def _ref(resource_type: str, row: dict[str, Any]) -> ContentCampaignInputRef:
        return ContentCampaignInputRef(
            resource_type=resource_type,
            resource_id=str(row["external_id"]),
            content_hash="sha256:" + str(row["payload_hash"]),
            authority="ecom_object",
        )

    @classmethod
    def _product(cls, row: dict[str, Any]) -> ContentCampaignProductInput:
        props = dict(row["properties"])
        title = str(props.get("title") or "").strip() or "商品名称来源未提供"
        return ContentCampaignProductInput(
            product_ref=cls._ref("Product", row),
            title=title,
            price=str(props["price"]) if props.get("price") is not None else None,
            available_stock=_non_negative_int(props.get("stock")),
            business_status=str(props["state"]) if props.get("state") is not None else None,
            source_updated_at=row["source_updated_at"],
        )

    @classmethod
    def _inventory(cls, row, products):
        props = dict(row["properties"])
        return ContentCampaignInventoryInput(
            sku_ref=cls._ref("ProductSku", row),
            product_ref=products.get(str(props.get("productId", ""))),
            stock=_non_negative_int(props.get("stock")),
            stock_alarm=_non_negative_int(props.get("stockAlarm")),
            stock_health=props.get("stock_health"),
            source_updated_at=row["source_updated_at"],
        )

    @staticmethod
    def _authority_ref(resource_type, observation, identity):
        return ContentCampaignInputRef(
            resource_type=resource_type,
            resource_id=identity,
            revision=observation.revision.revision,
            content_hash="sha256:" + observation.revision.content_hash,
            authority="content_campaign_authority",
        )

    @classmethod
    def _campaign(cls, observation):
        item = observation.revision
        return ContentCampaignPlanInput(
            plan_ref=cls._authority_ref("CampaignRevision", observation, item.campaign_id),
            lifecycle=item.lifecycle.value,
            channel_ids=item.channel_ids,
            source_updated_at=item.created_at,
        )

    @classmethod
    def _calendar(cls, observation):
        item = observation.revision
        return ContentCampaignPlanInput(
            plan_ref=cls._authority_ref("CalendarEntryRevision", observation, item.entry_id),
            lifecycle=item.lifecycle.value,
            channel_ids=[],
            local_start=item.local_start,
            local_end=item.local_end,
            source_updated_at=item.created_at,
        )

"""GET admission candidate over a current, signed, explicitly selected installation.

This adapter does not activate HTTP routes or grant runtime readiness. Legacy
admission is preserved; selected installed assets require actual exact scope grants.
"""
from __future__ import annotations

from aos_api.asset_registry.errors import AssetNotFoundError
from aos_api.ecommerce_workshop_contracts import WorkshopDependencyType


class EcommerceWorkshopReadCatalog:
    def __init__(self, *, legacy, installed=None, installed_factory=None) -> None:
        self._legacy = legacy
        self._installed = installed
        self._installed_factory = installed_factory

    def get_readiness(self, **scope):
        try:
            return self._legacy.get_readiness(**scope)
        except AssetNotFoundError:
            # Integrity, signature and availability errors never permit fallback.
            pass
        return self.get_installed_readiness(**scope)

    def get_installed_readiness(self, **scope):
        """Check only the selected installation with the existing exact read gate."""
        installed = (
            self._installed_factory() if self._installed_factory is not None
            else self._installed
        )
        result = installed.get_readiness(**scope)
        item = result.item
        installation = item.installation_ref
        required = set(item.permissions.data_scopes)
        expected_revision = (
            f'installation:{installation.revision}/lock:{installation.lock_revision}'
        )
        granted = {
            ref.resource_id for ref in item.dependency_refs
            if ref.resource_type == 'DataScopeGrant'
            and ref.revision == expected_revision
            and ref.content_hash == installation.lock_hash
            and ref.authority == 'postgres:bundle_composition_lock'
        }
        permitted_gap_types = {
            WorkshopDependencyType.OBJECT,
            WorkshopDependencyType.CAPABILITY,
            WorkshopDependencyType.AIP_FEATURE,
        }
        disallowed = any(
            blocker.dependency_type not in permitted_gap_types
            and not (
                blocker.dependency_type == WorkshopDependencyType.INSTALLATION
                and blocker.reason_code == 'INSTALLED_ASSET_READONLY'
                and blocker.dependency_id == installation.installation_id
            )
            for blocker in item.blockers
        )
        if 'ecommerce.workshop.read' not in required or not required <= granted or disallowed:
            raise AssetNotFoundError('Workshop module is not installed for scoped reading')
        # Keep the original disabled state and every runtime/authority gap intact.
        return result

                    "LEFT JOIN aip_artifact master ON master.org_id=relation.org_id "

                    "AND receipt.project_id=relation.project_id "
                    "AND receipt.operation='artifact_relation.create' "
                    "AND receipt.result_ref->>'resourceType'='ArtifactRelation' "
                    "AND receipt.result_ref->>'resourceId'=relation.relation_id "
                    "WHERE relation.org_id=%s AND relation.project_id=%s "
                    "AND relation.relation_type='variant_of' "

def _list_current_on_connection(
        self,
        conn: Any,
        scope: TenantScope,
        *,
        cutoff: datetime,
        limit: int,
        model: type[RevisionT],
        head_table: str,
        revision_table: str,
        identity_column: str,
    ) -> list[ContentCampaignAuthorityObservation[RevisionT]]:
        if cutoff.utcoffset() is None:
            raise ValueError("authority cutoff requires a timezone")
        if not 1 <= limit <= 100:
            raise ValueError("authority read limit must be between 1 and 100")
        try:
            rows = conn.execute(
                    f"SELECT revision.org_id,revision.project_id,"
                    f"revision.receipt_id,revision.authority_data FROM {head_table} head "
                    f"JOIN {revision_table} revision "
                    "ON revision.org_id=head.org_id "
                    "AND revision.project_id=head.project_id "
                    f"AND revision.{identity_column}=head.{identity_column} "
                    "AND revision.revision=head.current_revision "
                    "WHERE head.org_id=%s AND head.project_id=%s "
                    "AND revision.created_at<=%s "
                    f"ORDER BY revision.created_at DESC,revision.{identity_column} "
                    "LIMIT %s",
                    (*scope.key, cutoff, limit),
                ).fetchall()
            items = []
            for row in rows:
                if (row["org_id"], row["project_id"]) != scope.key:
                    raise ValueError("content-campaign authority tenant scope drift")
                item = model.model_validate(self._load(row["authority_data"]))
                self._require_item_scope(
                    scope, item.tenant.org_id, item.tenant.project_id
                )
                items.append(
                    ContentCampaignAuthorityObservation(
                        revision=item,
                        receipt_id=row["receipt_id"],
                    )
                )
            return items
        except ContentCampaignAuthorityConflict as exc:
            raise ContentCampaignAuthorityReadError(str(exc)) from exc
        except (psycopg.Error, KeyError, TypeError, ValueError) as exc:
            raise ContentCampaignAuthorityReadError(
                "content-campaign authority read failed closed"
            ) from exc

def _connect(
    scope: TenantScope | None = None,
    *,
    inherit_scope: bool = True,
    read_only: bool = False,
    isolation_level: IsolationLevel | None = None,
    platform_role: bool = False,
) -> Iterator[psycopg.Connection]:
    dsn = get_dsn()
    log.debug("db_connect host_port_from_env=%s", "AOS_DATABASE_URL" in os.environ)
    with psycopg.connect(dsn, row_factory=dict_row) as conn:
        # psycopg applies these characteristics when the transaction starts.
        # They must be configured before client encoding or tenant RLS executes
        # the first SQL statement; issuing SET TRANSACTION afterwards fails.
        if isolation_level is not None:
            conn.isolation_level = isolation_level
        if read_only:
            conn.read_only = True
        # 强制 UTF-8，避免客户端/驱动默认编码把中文写成 ???
        conn.execute("SET client_encoding TO 'UTF8'")
        effective_scope = (
            scope
            if scope is not None
            else (current_tenant_scope() if inherit_scope else None)
        )
        if effective_scope is not None:
            apply_transaction_scope(conn, effective_scope)
        elif platform_role:
            # Scope-less but still authenticated as the runtime role: needed on
            # a NOINHERIT login where omitting SET ROLE means "no privileges at
            # all". Never grants tenant visibility — the GUCs stay unset.
            apply_platform_role(conn)
        else:
            # Scope-less AND no platform role: on a NOINHERIT login this
            # connection can read nothing at all, and the failure surfaces as
            # "permission denied" deep inside whichever statement runs first.
            # Attribute the caller so the residual offenders can be census'd
            # instead of guessed (see W6 connection-plane debt work).
            try:
                frame = sys._getframe(2)
                caller = f"{frame.f_code.co_filename.rsplit('/', 1)[-1]}:{frame.f_lineno}"
            except ValueError:  # shallow stack — never let diagnostics break a connect
                caller = "unknown"
            log.debug("db_connect_without_scope caller=%s", caller)
        yield conn
def connect_read_only(
