export function parseThreeModuleClosureContributionView(value: unknown, expectedModule?: ThreeModuleId, expectedTenant?: WorkshopTenant): ThreeModuleClosureContributionViewResponse { const raw = record(value, "threeModule"); exact(raw, ["schemaVersion", "tenant", "module", "evaluatedAt", "atomicSkillIds", "logicId", "primaryColleague", "collaboratorColleagues", "latestClosure", "latestUsage", "latestEffect", "latestHandoff", "blockers", "allowedCommands", "handoffConsumeAllowed", "memoryPromotionAllowed", "externalEffectsAllowed"], "threeModule"); const tenant = parseTenant(raw.tenant); const module = enumValue(raw.module, THREE_MODULE_IDS, "threeModule.module"); if (raw.schemaVersion !== THREE_MODULE_CLOSURE_SCHEMA_VERSION || (expectedModule && module !== expectedModule) || (expectedTenant && (tenant.orgId !== expectedTenant.orgId || tenant.projectId !== expectedTenant.projectId)) || !Array.isArray(raw.atomicSkillIds) || !raw.atomicSkillIds.length || !Array.isArray(raw.collaboratorColleagues) || !Array.isArray(raw.blockers) || !Array.isArray(raw.allowedCommands) || (raw.allowedCommands.length !== 0 && (raw.allowedCommands.length !== THREE_MODULE_COMMANDS.length || !raw.allowedCommands.every((command, index) => command === THREE_MODULE_COMMANDS[index]))) || raw.handoffConsumeAllowed !== false || raw.memoryPromotionAllowed !== false || raw.externalEffectsAllowed !== false) throw new TypeError("threeModule contract 漂移"); const serialized = JSON.stringify(raw).toLowerCase(); for (const forbidden of ["mobile", "openid", "email", "address", "contactvalue", "providerpayload"]) if (serialized.includes(forbidden)) throw new TypeError("threeModule 泄露 PII/provider payload"); return { schemaVersion: THREE_MODULE_CLOSURE_SCHEMA_VERSION, tenant, module, evaluatedAt: timestamp(raw.evaluatedAt, "threeModule.evaluatedAt"), atomicSkillIds: raw.atomicSkillIds.map((item) => boundedText(item, "threeModule.skill", 200)), logicId: boundedText(raw.logicId, "threeModule.logicId", 200), primaryColleague: boundedText(raw.primaryColleague, "threeModule.primary", 80), collaboratorColleagues: raw.collaboratorColleagues.map((item) => boundedText(item, "threeModule.collaborator", 80)), latestClosure: raw.latestClosure === null ? null : parseThreeModuleClosure(raw.latestClosure, tenant, module), latestUsage: raw.latestUsage === null ? null : parseThreeModuleUsage(raw.latestUsage, tenant), latestEffect: raw.latestEffect === null ? null : parseThreeModuleEffect(raw.latestEffect, tenant), latestHandoff: raw.latestHandoff === null ? null : parseThreeModuleHandoff(raw.latestHandoff, tenant), blockers: raw.blockers.map((item) => boundedText(item, "threeModule.blocker", 200)), allowedCommands: raw.allowedCommands.length === 0 ? [] : [...THREE_MODULE_COMMANDS], handoffConsumeAllowed: false, memoryPromotionAllowed: false, externalEffectsAllowed: false }; }
