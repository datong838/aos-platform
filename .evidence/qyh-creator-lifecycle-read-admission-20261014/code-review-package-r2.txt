def lifecycle_admission(*, readonly=True):
    from tests.test_ecommerce_workshop_read_catalog import make, ARGS
    catalog, _ = make()
    result = catalog.get_readiness(**{**ARGS, 'module_id': 'ecommerce.creator-growth'})

def lifecycle_probes(client, *, failure=None):
    calls = []
    class Legacy:
        def view(self, scope):
            calls.append(('legacy', scope.key))
            raise ReadReached('legacy-lifecycle')
    class Readonly:
        def read(self, scope):
            calls.append(('readonly', scope.key))
            if failure:
                from aos_api.ecommerce_workshop_creator_lifecycle_readonly import CreatorLifecycleReadonlyError
                raise CreatorLifecycleReadonlyError(failure)
            raise ReadReached('readonly-lifecycle')
    client.app.dependency_overrides[routes.get_ecommerce_workshop_creator_lifecycle] = lambda: Legacy()
    client.app.dependency_overrides[routes.get_creator_lifecycle_readonly] = lambda: Readonly()
    return calls

ACTUAL FUNCTION get_creator_lifecycle_view
def get_creator_lifecycle_view(
    request: Request,
    principal: PrincipalDependency,
    catalog: ReadCatalogDependency,
    service: CreatorLifecycleDependency,
    readonly: CreatorLifecycleReadonlyDependency,
) -> CreatorLifecycleView:
    _reject_query_parameters(request)
    admission = _require_creator_growth_installation(principal=principal, catalog=catalog)
    if not isinstance(admission, EcommerceWorkshopModuleReadinessResponse):
        raise ApiError(
            code="CREATOR_LIFECYCLE_READ_ADMISSION_UNAVAILABLE",
            message="当前生命周期读取准入不可核验。",
            status_code=503,
        )
    scope = TenantScope(principal.org_id, principal.project_id)
    if any(blocker.reason_code == "INSTALLED_ASSET_READONLY" for blocker in admission.item.blockers):
        try:
            return readonly.read(scope)
        except CreatorLifecycleReadonlyError as exc:
            raise ApiError(
                code=exc.code,
                message="当前生命周期数据暂不可读，请核对数据或读取范围。",
                status_code=503,
            ) from exc
    return service.view(scope)
